# A MASC-format corpus written for the turbo-formats tests: hand-written text
# and stand-off annotation in the layout of MASC 3.0 (GrAF). Not MASC data.
# Sentence regions: overlaps, touching and empty regions, and a region past the text.

      <annotation loc="s-penn.xml" f.id="f.penn">annotation</annotation>
    </annotations>
  </profileDesc>
</documentHeader>
