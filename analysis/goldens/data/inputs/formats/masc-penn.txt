# A MASC-format corpus written for the turbo-formats tests: hand-written text
# and stand-off annotation in the layout of MASC 3.0 (GrAF). Not MASC data.
# Penn tokens: shared quarks, tokens across sentences, missing tags and empty tokens.

@@ file penn.hdr
<?xml version="1.0" encoding="UTF-8"?>
<documentHeader xmlns="http://www.xces.org/ns/GrAF/1.0/" docId="TEST" version="1.0.4">
  <profileDesc>
    <primaryData f.id="f.text" loc="penn.txt"/>
    <annotations>
      <annotation loc="penn-s.xml" f.id="f.s">annotation</annotation>
      <annotation loc="penn-seg.xml" f.id="f.seg">annotation</annotation>
      <annotation loc="penn-penn.xml" f.id="f.penn">annotation</annotation>
    </annotations>
  </profileDesc>
</documentHeader>

@@ end
