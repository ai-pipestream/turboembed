# A MASC-format corpus written for the turbo-formats tests: hand-written text
# and stand-off annotation in the layout of MASC 3.0 (GrAF). Not MASC data.
# Documents OpenNLP logs and skips: each fails with an IOException from parseDocument.

<?xml version="1.0" encoding="UTF-8"?>
<documentHeader xmlns="http://www.xces.org/ns/GrAF/1.0/" docId="TEST" version="1.0.4">

    <primaryData f.id="f.text" loc="good.txt"/>
    <annotations>
      <annotation loc="good-s.xml" f.id="f.s">annotation</annotation>
      <annotation loc="good-seg.xml" f.id="f.seg">annotation</annotation>
      <annotation loc="good-penn.xml" f.id="f.penn">annotation</annotation>
