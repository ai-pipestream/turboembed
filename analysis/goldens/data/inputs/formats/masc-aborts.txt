# A MASC-format corpus written for the turbo-formats tests: hand-written text
# and stand-off annotation in the layout of MASC 3.0 (GrAF). Not MASC data.
# Documents that end MascDocumentStream's constructor: the first of them aborts the stream.

<?xml version="1.0" encoding="UTF-8"?>
<documentHeader xmlns="http://www.xces.org/ns/GrAF/1.0/" docId="TEST" version="1.0.4">
  <profileDesc>
    <primaryData f.id="f.text" loc="first.txt"/>
    <annotations>
      <annotation loc="first-s.xml" f.id="f.s">annotation</annotation>
      <annotation loc="first-seg.xml" f.id="f.seg">annotation</annotation>
    </annotations>
  </profileDesc>
</documentHeader>

@@ end
